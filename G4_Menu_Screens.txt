
static void drawFallingLetters(float alpha) {
    for (int i = 0; i < 34; i++) {
        float x = (float)((i * 97) % 1280);
        float y = fmodf(clockT * (14.0f + (float)(i % 4) * 8.0f) + (float)(i * 61), 760.0f) - 20.0f;
        char ch[2] = { (char)('A' + (i * 7) % 26), 0 };
        DrawText(ch, (int)x, (int)y, 22 + (i % 3) * 6, Fade(C_GOLD, alpha));
    }
}

static void screenNew() {
    drawFallingLetters(0.08f);
    Rectangle box = { 340, 170, 600, 380 };
    panel(box, "NEW GAME");
    textC("Enter your detective name", 640, 230, 26, C_TEXT);

    // typing
    int ch = GetCharPressed();
    while (ch > 0) {
        if (ch >= 32 && ch < 127 && playerName.size() < 14)
            playerName += (ch == ' ') ? '_' : (char)ch;   // spaces become underscores
        ch = GetCharPressed();
    }
    if ((IsKeyPressed(KEY_BACKSPACE) || IsKeyPressedRepeat(KEY_BACKSPACE)) && !playerName.empty())
        playerName.erase(playerName.size() - 1);

    Rectangle field = { 400, 280, 480, 56 };
    DrawRectangleRounded(field, 0.2f, 8, C_BG);
    DrawRectangleLinesEx(field, 2, C_GOLD);
    string shown = playerName;
    if (((int)(clockT * 2)) % 2 == 0)
        shown += "|";
    DrawText(shown.c_str(), 416, 296, 28, C_TEXT);
    textC("Spaces turn into underscores (_).  Date: " + todayDate, 640, 350, 16, C_DIM);
    if (!menuError.empty())
        textC(menuError, 640, 380, 18, C_RED);

    bool go = button(Rectangle{ 400, 430, 230, 58 }, "Start", C_GREEN, true, 24) || IsKeyPressed(KEY_ENTER);
    if (go) {
        if (playerName.empty()) {
            pushToast("Name cannot be empty.", true);
            menuError = "Name cannot be empty.";
        } else {
            startGame();
        }
    }
    if (button(Rectangle{ 650, 430, 230, 58 }, "Back", C_PANEL2, true, 24) || IsKeyPressed(KEY_ESCAPE))
        goScreen(SCR_MENU);
}

static int splitBar(const string& line, string out[], int maxF) {
    int n = 0;
    string cur = "";
    bool started = false;
    for (int i = 0; i < (int)line.size(); i++) {
        if (line[i] == '|') {
            if (started && n < maxF) {
                int a = 0, b = (int)cur.size();
                while (a < b && cur[a] == ' ') a++;
                while (b > a && cur[b - 1] == ' ') b--;
                out[n++] = cur.substr(a, b - a);
            }
            started = true;
            cur = "";
        } else {
            cur += line[i];
        }
    }
    return n;
}

static void screenLeader() {
    drawFallingLetters(0.06f);
    Rectangle box = { 60, 50, 1160, 620 };
    panel(box, "LEADER BOARD");
    string rows[200];
    int n = 0;
    try {
        n = eng->getLeaderBoard()->getRows(rows, 200);
    } catch (GameException& e) {
        textC(e.what(), 640, 300, 24, C_RED);
    }
    float xs[6] = { 90, 360, 500, 650, 800, 1090 };
    const char* heads[6] = { "Detective", "Result", "Credibility", "Turns left", "Culprit", "Date" };
    for (int i = 0; i < 6; i++)
        DrawText(heads[i], (int)xs[i], 96, 20, C_GOLD);
    DrawRectangle(80, 124, 1120, 2, C_GOLD);
    int shownRows = 0;
    for (int r = n - 1; r >= 1 && shownRows < 15; r--) {
        string f[6];
        if (splitBar(rows[r], f, 6) != 6)
            continue;
        float y = 136.0f + shownRows * 32.0f;
        if (shownRows % 2 == 0)
            DrawRectangle(80, (int)y - 2, 1120, 30, Color{ 36, 42, 70, 255 });
        for (int i = 0; i < 6; i++) {
            Color c = C_TEXT;
            if (i == 1)
                c = (f[1] == "WIN") ? C_GREEN : C_RED;
            DrawText(pretty(f[i]).c_str(), (int)xs[i], (int)y + 3, 20, c);
        }
        shownRows++;
    }
    if (shownRows == 0)
        textC("No games played yet. Solve a case first!", 640, 300, 26, C_DIM);
    if (button(Rectangle{ 540, 612, 200, 44 }, "Back", C_PANEL2, true, 22) || IsKeyPressed(KEY_ESCAPE))
        goScreen(SCR_MENU);
}
